SELECT c.course_id AS id, c.name, ci.img_url AS thumbnailUrl, c.distance, c.duration, c.max_ele AS maxElevation, (ST_Distance_Sphere(c.start_point, ST_PointFromText('POINT(35.1587 129.1604)', 4326)) / 1000) AS distanceFromUser FROM course_exp c IGNORE INDEX (idx_exp_location, idx_exp_lat_lon_v, idx_exp_lat_lon_s) LEFT JOIN course_image_exp ci ON c.course_id = ci.course_id WHERE c.start_lat_s BETWEEN 35.1132842 AND 35.2041158 AND c.start_lon_s BETWEEN 129.1048495 AND 129.2159505 AND ST_Distance_Sphere(c.start_point, ST_PointFromText('POINT(35.1587 129.1604)', 4326)) <= 5000 ORDER BY distanceFromUser ASC;
